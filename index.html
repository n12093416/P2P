import asyncio import json import os import websockets # 연결된 접속자(피어) 목록 ROOM = set() async def handler(websocket): ROOM.add(websocket) print(f"새 피어 연결: {websocket.remote_address}") try: async for message in websocket: data = json.loads(message) # 나를 제외한 다른 피어에게 시그널링 데이터 전달 for client in ROOM: if client != websocket: await client.send(json.dumps(data)) except websockets.exceptions.ConnectionClosedError: pass finally: ROOM.remove(websocket) print(f"피어 연결 종료: {websocket.remote_address}") async def main(): # Render의 PORT 환경 변수를 읽어옴 (기본값 8765) port = int(os.environ.get("PORT", 8765)) # 외부 접속 허용을 위해 0.0.0.0으로 바인딩 async with websockets.serve(handler, "0.0.0.0", port): print(f"시그널링 서버 실행 중 (Port: {port})...") await asyncio.Future() if __name__ == "__main__": asyncio.run(main())
